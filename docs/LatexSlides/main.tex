\section{Motivation and Research Problem}

\begin{frame}{Why this matters in Digital Forensics}
\begin{columns}
\begin{column}{0.54\textwidth}
\begin{itemize}
    \item AI-based forensic tools are increasingly used to \highlight{triage large image collections}.
    \item The analyst often relies on labels, tags, bookmarks or semantic retrieval outputs.
    \item In forensic contexts, a missed item is not only a technical error: it can become an \highlight{operational risk}.
\end{itemize}
\end{column}
\begin{column}{0.40\textwidth}
\begin{colorblock}[black]{sinteflightgreen}{Core message}
The contribution is not a new weapon classifier.\\[0.35em]
Proxy models are controlled experimental references used to evaluate operational robustness.
\end{colorblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Research questions}
\begin{block}{Main question}
How robust are AI-based forensic image-classification and media-triage systems when exposed to adversarial, anti-forensic and out-of-distribution inputs?
\end{block}

\small
\begin{enumerate}
    \item \highlight{RQ1}: Proxy robustness across clean, OOD, adversarial and anti-forensic conditions.
    \item \highlight{RQ2}: Direct-target attacks, cross-architecture transfer and model-agnostic Color Shift.
    \item \highlight{RQ3}: Black-box behavior, observable output semantics and configuration sensitivity.
    \item \highlight{RQ4}: Operational error patterns and implications for continued human review.
\end{enumerate}
\end{frame}

\begin{frame}{Research gap}
\begin{itemize}
    \item Existing evaluations often emphasize clean performance.
    \item Commercial forensic tools expose heterogeneous and partial decision signals.
    \item Robustness under anti-forensic image transformations is rarely assessed with an operational protocol.
    \item OOD behavior is often mixed with classification performance, instead of being analyzed separately.
\end{itemize}

\begin{colorblock}[black]{sintefyellow}{Gap addressed by this thesis}
A task-specific methodological protocol that tests clean, OOD, adversarial and anti-forensic conditions under a common traceability logic.
\end{colorblock}
\end{frame}

\section{Experimental Design}

\begin{frame}{Dataset construction}
\begin{columns}
\begin{column}{0.50\textwidth}
\begin{block}{Frozen source dataset}
\begin{itemize}
    \item 1,500 total images.
    \item 500 weapon.
    \item 500 non-weapon.
    \item 500 out-of-distribution.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.46\textwidth}
\begin{block}{Binary subset}
\begin{itemize}
    \item 1,000 images.
    \item 500 weapon.
    \item 500 non-weapon.
    \item Base for proxy and perturbation generation.
\end{itemize}
\end{block}
\end{column}
\end{columns}

\begin{colorblock}[black]{sintefyellow}{Important distinction}
OOD images are not a third class. They are used to test how systems behave outside the intended distribution.
\end{colorblock}
\end{frame}

\begin{frame}{Forensic evaluation bundle}
\begin{table}
\centering
\small
\begin{tabular}{@{}lr@{}}
\toprule
\textbf{Bundle component} & \textbf{Files} \\
\midrule
Clean binary images & 1,000 \\
Clean OOD images & 500 \\
Adversarial samples & 5,000 \\
Anti-forensic samples & 5,000 \\
\midrule
\textbf{Total} & \textbf{11,500} \\
\bottomrule
\end{tabular}
\end{table}

\begin{block}{Why this matters}
The bundle is processed blindly by forensic tools; ground truth and evaluation metadata remain outside the tool-facing view. The 10,000 manipulated files are source--condition variants, not independent real-world cases.
\end{block}
\end{frame}

\begin{frame}[shrink=2]{Workflow}
\small
\begin{enumerate}
    \item Source image pool.
    \item Human-in-the-loop validation and dataset freezing.
    \item Binary subset and clean OOD split.
    \item Adversarial and anti-forensic perturbation generation.
    \item Blind bundle construction.
    \item Proxy-model evaluation.
    \item Commercial black-box evaluation.
    \item Output normalization, metrics, XAI and audit controls.
\end{enumerate}

\begin{colorblock}[black]{sinteflightgreen}{Traceability principle}
FAIRLab separates source data, transformations, tool-facing inputs and evaluation metadata while preserving deterministic identifiers and checksums.
\end{colorblock}
\end{frame}

\begin{frame}{Proxy models vs black-box systems}
\begin{columns}
\begin{column}{0.47\textwidth}
\begin{block}{Transparent proxies}
\begin{itemize}
    \item EfficientNet-B0.
    \item ResNet18.
    \item CLIP: frozen visual encoder + supervised binary head.
    \item Controlled training, gradients and diagnostic outputs.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Black-box systems}
\begin{itemize}
    \item Magnet AXIOM / Magnet.AI.
    \item Excire Foto 2025: D20, D50 and D80.
    \item Cellebrite Inseyets.
    \item Magnet Griffeye / T3K CORE.
\end{itemize}
\end{block}
\end{column}
\end{columns}

\begin{block}{Methodological point}
Proxy models provide experimental control; four software systems and six frozen configurations provide operational realism.
\end{block}
\end{frame}

\begin{frame}{Perturbation suite}
\small
\begin{columns}
\begin{column}{0.48\textwidth}
\begin{block}{Adversarial branch}
\begin{itemize}
    \item FGSM.
    \item One Pixel Attack.
    \item Sigma Zero Attack.
    \item SuperDeepFool.
    \item Color Shift: model-agnostic adversarial-style condition.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Anti-forensic branch}
\begin{itemize}
    \item JPEG Recompression.
    \item Resample + Resize.
    \item Gaussian Blur.
    \item Histogram Modification.
    \item Contrast Stretching.
\end{itemize}
\end{block}
\end{column}
\end{columns}

\begin{block}{Direct target vs transfer}
Model-dependent attacks were generated only against EfficientNet-B0: direct-target evaluation on EfficientNet-B0 and empirical transfer evaluation on ResNet18, CLIP and the black-box systems.
\end{block}
\end{frame}

\section{Evaluation Logic}

\begin{frame}{Output normalization}
Commercial tools do not expose the same kind of output.

\begin{table}
\centering
\scriptsize
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Tool} & \textbf{Observable forensic signal} \\
\midrule
Magnet AXIOM / Magnet.AI & Possible weapons tag \\
Excire Foto 2025 & Union of eight firearm-oriented semantic queries \\
Cellebrite Inseyets & Exported \texttt{Armi} / \texttt{Pistola} / \texttt{Fucile} labels \\
Griffeye / T3K CORE & CORE/Violence/Firearm bookmark \\
\bottomrule
\end{tabular}
\end{table}

\begin{colorblock}[black]{sintefyellow}{Critical point}
The thesis evaluates observable exported signals, not proprietary model internals, thresholds, weights or training data.
\end{colorblock}
\end{frame}

\begin{frame}[shrink=2]{Metrics: not only accuracy}
\small
\begin{columns}
\begin{column}{0.48\textwidth}
\begin{block}{Binary performance}
\begin{itemize}
    \item Accuracy and balanced accuracy.
    \item Weapon recall, precision and F1.
    \item False-negative and false-positive rates.
\end{itemize}
\end{block}
\begin{block}{Robustness}
\begin{itemize}
    \item Accuracy drop and induced errors.
    \item Directional prediction transitions.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{OOD and coverage}
\begin{itemize}
    \item OOD weapon-assignment rate.
    \item Matching and unknown-state handling.
\end{itemize}
\end{block}
\begin{block}{Proxy-only diagnostics}
\begin{itemize}
    \item Maximum predicted-class probability.
    \item Qualitative Integrated Gradients analysis.
\end{itemize}
\end{block}
\end{column}
\end{columns}

\begin{block}{Forensic framing}
The central risk is not only aggregate error, but the direction of errors and whether relevant evidence is silently deprioritized.
\end{block}
\end{frame}

\section{Results and Interpretation}

\begin{frame}{Proxy-model results}
\begin{table}
\centering
\scriptsize
\begin{tabular}{@{}lccc@{}}
\toprule
\textbf{Model} & \textbf{Clean acc.} & \textbf{Largest adversarial drop} & \textbf{Largest anti-forensic drop} \\
\midrule
EfficientNet-B0 & 0.978 & 0.963 -- Sigma Zero (direct) & 0.013 -- Gaussian Blur \\
ResNet18 & 0.964 & 0.046 -- FGSM (transfer) & 0.030 -- Histogram Mod. \\
CLIP proxy & 0.927 & 0.013 -- FGSM (transfer) & 0.002 -- JPEG / Histogram \\
\bottomrule
\end{tabular}
\end{table}

\begin{itemize}
    \item EfficientNet-B0 falls from 0.978 to 0.015 under direct-target Sigma Zero.
    \item The same frozen artifacts do not produce equivalent collapse on ResNet18 or CLIP.
    \item Anti-forensic transformations produce smaller aggregate degradation, but still induce operationally relevant error transitions.
\end{itemize}

\begin{colorblock}[black]{sinteflightgreen}{Main result}
Clean performance does not imply operational robustness, and direct-target vulnerability must be distinguished from transferability.
\end{colorblock}
\end{frame}

\begin{frame}[shrink=3]{Black-box results}
\begin{table}
\centering
\scriptsize
\begin{tabular}{@{}lrrrr@{}}
\toprule
\textbf{Configuration} & \textbf{Accuracy} & \textbf{Recall} & \textbf{FNR} & \textbf{FPR} \\
\midrule
Magnet AXIOM / Magnet.AI & 0.933 & 0.901 & 0.099 & 0.035 \\
Excire D20 & 0.911 & 0.857 & 0.143 & 0.036 \\
Excire D50 & 0.925 & 0.949 & 0.051 & 0.099 \\
Excire D80 & 0.887 & 0.981 & 0.019 & 0.207 \\
Cellebrite Inseyets & 0.958 & 0.958 & 0.042 & 0.042 \\
Griffeye / T3K CORE & 0.972 & 0.951 & 0.049 & 0.007 \\
\bottomrule
\end{tabular}
\end{table}

\begin{itemize}
    \item Excire shows the clearest configuration-sensitive trade-off between missed targets and review workload.
    \item No unknown normalized decisions occurred in the canonical 69,000-decision population.
\end{itemize}

\begin{block}{Interpretation boundary}
These are protocol- and configuration-specific operational profiles, not a universal ranking or a reverse engineering of proprietary models.
\end{block}
\end{frame}

\begin{frame}{OOD behavior}
\begin{itemize}
    \item OOD images probe behavior outside the nominal weapon / non-weapon task.
    \item An OOD weapon assignment is not a supervised false positive, but it may increase review workload or distort prioritization.
    \item Excire illustrates configuration sensitivity: the OOD weapon-assignment rate rises from 0.238 at D20 to 0.522 at D80.
    \item Proxy Max-P values are intra-model diagnostic indicators only; they are not calibrated probabilities comparable across architectures.
\end{itemize}

\begin{colorblock}[black]{sintefyellow}{Denominator caveat}
Proxy OOD rates aggregate five fold-specific responses per image; black-box rates use one normalized decision per image.
\end{colorblock}
\end{frame}

\begin{frame}{Explainability and case studies}
\begin{itemize}
    \item Integrated Gradients is applied to five purposively selected EfficientNet-B0 cases.
    \item The analysis is qualitative and diagnostic, not a quantitative robustness metric.
    \item It compares model focus under clean and perturbed conditions.
    \item It does not explain the proprietary black-box systems and is not treated as legal proof.
\end{itemize}

\begin{block}{Message for the commission}
Explainability supports interpretation and debugging, but it does not replace validation, traceability or human forensic review.
\end{block}
\end{frame}

\section{Forensic Implications}

\begin{frame}{Operational implications}
\begin{itemize}
    \item AI-based forensic tools are useful for prioritization and workload reduction.
    \item They should not be used as autonomous evidence selectors.
    \item Every AI-based triage result should be linked to tool version, configuration, export logic and validation context.
    \item Human review remains necessary, especially for negative decisions, OOD assignments and configuration-sensitive outputs.
\end{itemize}
\end{frame}

\begin{frame}{Limitations}
\small
\begin{itemize}
    \item The corpus contains 1,500 manually selected images; review was performed by one reviewer without inter-rater agreement.
    \item Each perturbation condition uses one frozen parameterization rather than a severity sweep.
    \item Model-dependent attacks were generated only against EfficientNet-B0.
    \item Black-box findings depend on software version, configuration, export semantics and normalization mapping.
    \item The study evaluates operational triage behavior, not legal admissibility or evidentiary weight.
\end{itemize}

\begin{colorblock}[black]{sinteflightgreen}{Scope of the claims}
The thesis provides a protocol-specific robustness assessment, not software certification, universal ranking or evidentiary validation.
\end{colorblock}
\end{frame}

\begin{frame}{Conclusions}
\begin{enumerate}
    \item FAIRLab provides a reproducible and auditable protocol for forensic AI robustness assessment.
    \item Clean accuracy alone is insufficient for evaluating operational reliability.
    \item Direct-target collapse does not imply equivalent cross-architecture or black-box transfer.
    \item Black-box behavior depends on observable output semantics and documented configuration.
    \item OOD assignments and directional errors reinforce the need for validation-driven, human-supervised triage.
\end{enumerate}
\end{frame}

\begin{frame}{Future work}
\begin{itemize}
    \item Larger, more diverse forensic corpora with multi-annotator validation and agreement measures.
    \item Matched direct-target attacks against all transparent proxy architectures.
    \item OOD rejection, abstention and open-set mechanisms evaluated against analyst workload.
    \item Multiple transformation severity levels and chained adversarial / anti-forensic conditions.
    \item Longitudinal testing across software versions, configurations and defensible normalization mappings.
\end{itemize}
\end{frame}
