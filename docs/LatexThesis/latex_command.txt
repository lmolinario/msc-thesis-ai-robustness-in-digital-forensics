# Preferred cross-platform build
latexmk -pdf -interaction=nonstopmode -file-line-error main.tex

# Equivalent explicit sequence
pdflatex main.tex
makeglossaries main
biber main
pdflatex main.tex
pdflatex main.tex

# Glossary fallback when makeglossaries is unavailable
makeindex -s main.ist -t main.alg -o main.acr main.acn

# PowerShell example for a portable local Perl installation.
# Set the location through an environment variable instead of committing a
# workstation-specific absolute path.
$perlRoot = $env:STRAWBERRY_PERL_HOME
if ($perlRoot) {
    $env:Path = "$perlRoot\perl\bin;$perlRoot\c\bin;$env:Path"
}

# Remove auxiliary files on Unix-like systems
rm -f main.aux main.bbl main.bcf main.blg main.run.xml main.out \
main.toc main.lof main.lot main.acn main.acr main.alg \
main.glg main.glo main.gls main.ist main.glsdefs main.log

# Remove auxiliary files in PowerShell
Remove-Item -Path main.aux,main.bbl,main.bcf,main.blg,main.run.xml,main.out,main.toc,main.lof,main.lot,main.acn,main.acr,main.alg,main.glg,main.glo,main.gls,main.ist,main.glsdefs,main.log -Force -ErrorAction SilentlyContinue

# latexmk cleanup
latexmk -C
